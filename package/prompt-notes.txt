---

Crie uma aba principal "Notes" no dashboard (dashboard.html), ao lado de Tasks e Memory: notas livres em texto rico, gravadas num arquivo próprio, notes.json, na mesma pasta do tasks.csv.

Decisões de produto (não mudar):
- Layout: à esquerda, uma coluna de cards no estilo das colunas do Grid (reaproveita .matrix-chip, .matrix-add, .matrix-col-title/.matrix-col-sub, .matrix-chip-del, .matrix-chip-close). Clicar no card abre um editor grande à direita.
- Editor: título + corpo contenteditable com barra de formatação (negrito, itálico, sublinhado, riscado, H1/H2, texto normal, listas, citação, código, link, remover link, limpar formatação). Autosave 600 ms depois de cada edição.
- Links: botão 🔗 ou ⌘K/Ctrl+K pede a URL (com texto selecionado vira link no texto; sem seleção insere a URL). URL digitada vira link ao apertar espaço/Enter. Colar uma URL com texto selecionado transforma o texto em link. No editor, clique simples edita e ⌘/Ctrl+clique abre o link em nova aba. Nos cards e no painel, os links são clicáveis.
- Sanitização: HTML colado e HTML lido do arquivo passam por sanitizeNoteHtml: só formatação básica e links http/https/mailto; scripts, estilos e atributos saem. Texto puro colado tem as URLs convertidas em links.
- Nota vazia (sem título, sem texto e sem tasks vinculadas) é apagada sozinha ao sair dela. Excluir uma nota com conteúdo pede confirmação.
- Arrastar card reordena a coluna. "+ nova nota" fica no topo da coluna e cria a nota no topo.
- Busca (campo no cabeçalho da coluna): filtra a coluna e o painel; procura no título, no texto e nas tasks vinculadas (#id e título), sem acento e sem caixa, exigindo todas as palavras; Esc limpa.
- Vínculo nota -> task: fica SÓ no notes.json (campo `tasks` da nota = lista de ids da coluna `id` do tasks.csv). Nada é gravado no tasks.csv. No editor, barra "Tasks:" com chips na cor da frente (× desvincula; clicar abre a task no Grid em edição, ou mostra aviso se os filtros do Grid a escondem) e "+ vincular task" (busca por id ou título, até 10 resultados, não-Done primeiro e depois por score, ↑↓ Enter Esc). Tasks ainda sem id (recém-criadas, antes de gravar o tasks.csv) não aparecem na busca. Chips também nos cards e no painel; id que sumiu do CSV aparece como "#id (não encontrada)".
- Modo painel: sem nota aberta, a área do editor mostra um card-resumo por nota em grade (título, até 8 linhas do texto mantendo quebras, tasks, data, nº de links) e um card "+ nova nota" (some durante a busca). Botão "▦ Painel" no cabeçalho da coluna (laranja = painel visível): com nota aberta mostra o painel; no painel volta para a última nota aberta.
- Formato do notes.json (UTF-8, indentado com 2 espaços): { "version": 1, "notes": [{ "id", "title", "html", "tasks", "created", "updated" }] }. Ordem do array = ordem da coluna. Datas em ISO. O arquivo só é criado na primeira gravação (abrir a aba não cria nada). Também aceita na leitura um array puro de notas e notas sem `tasks`.
- Mudança por fora no notes.json recarrega a aba (checagem a cada 1 s), exceto com o foco dentro do editor.
- localStorage: `mainTab` (aba principal; restaura notes ou memory ao recarregar) e `notesOpen` (nota aberta).

    async function loadNotes(root) {
      notesDir = root;
      notesFileHandle = null;
      notes = [];
      notesLastModified = 0;
      try {
        notesFileHandle = await root.getFileHandle(NOTES_FILE);
        const file = await notesFileHandle.getFile();
        notesLastModified = file.lastModified;
        notes = parseNotes(await file.text());

        if (e.name !== 'NotFoundError') showStatus(NOTES_FILE + ': ' + e.message);
      }

      renderNotes();

    function parseNotes(text) {
      if (!text.trim()) return [];
      const data = JSON.parse(text);
      return (Array.isArray(data) ? data : data.notes || []).map(n => ({
        id: String(n.id), title: n.title || '', html: sanitizeNoteHtml(n.html || ''),
        tasks: Array.isArray(n.tasks) ? n.tasks.map(String) : [], created: n.created || '', updated: n.updated || n.created || '',

          notesLastModified = (await notesFileHandle.getFile()).lastModified;
          showStatus('Saved');
        } catch (e) {
          notesDirty = true;
          showStatus('Save failed: ' + e.message);
        }
      });

    }

        notes = parseNotes(await file.text());
        renderNotes();
        showStatus('Reloaded ' + NOTES_FILE);
      } catch (e) { console.log('Notes watch error:', e); }

    // Só formatação básica e links sobrevivem (colagem de páginas web, arquivo editado por fora)
    const NOTE_TAGS = new Set(['A', 'B', 'STRONG', 'I', 'EM', 'U', 'S', 'P', 'DIV', 'BR', 'UL', 'OL', 'LI', 'H1', 'H2', 'H3', 'BLOCKQUOTE', 'PRE', 'CODE', 'HR', 'SPAN']);
    function sanitizeNoteHtml(html) {
      const doc = new DOMParser().parseFromString('<div>' + html + '</div>', 'text/html');
      const clean = node => {
        [...node.childNodes].forEach(child => {
          if (child.nodeType === Node.TEXT_NODE) return;
          if (child.nodeType !== Node.ELEMENT_NODE || ['SCRIPT', 'STYLE', 'META', 'LINK', 'TITLE'].includes(child.tagName)) { child.remove(); return; }
          clean(child);
          if (!NOTE_TAGS.has(child.tagName)) { child.replaceWith(...child.childNodes); return; }
          const href = child.tagName === 'A' ? child.getAttribute('href') || '' : '';
          [...child.attributes].forEach(a => child.removeAttribute(a.name));
          if (child.tagName === 'A') {
            if (/^(https?:|mailto:)/i.test(href)) child.setAttribute('href', href);
            else child.replaceWith(...child.childNodes);
          }
        });
      };
      const root = doc.body.firstChild;
      clean(root);
      return root.innerHTML;
    }

      function insertNoteLink() {
        const sel = window.getSelection();
        const range = sel.rangeCount ? sel.getRangeAt(0) : null;
        const current = range && range.startContainer.parentElement?.closest('a');
        let url = prompt('URL do link:', current ? current.getAttribute('href') : 'https://');
        if (!url || url === 'https://') return;
        url = url.trim();
        if (!/^(https?:|mailto:)/i.test(url)) url = 'https://' + url;
        if (range) { sel.removeAllRanges(); sel.addRange(range); }

      // URL digitada vira link ao apertar espaço/Enter
      function autolinkBeforeCaret() {
        const sel = window.getSelection();
        if (!sel.rangeCount || !sel.isCollapsed) return;
        const node = sel.anchorNode;
        if (!node || node.nodeType !== Node.TEXT_NODE || node.parentElement.closest('a')) return;
        const before = node.data.slice(0, sel.anchorOffset);
        const m = before.match(/(?:https?:\/\/|www\.)\S+$/i);
        if (!m) return;
        const start = before.length - m[0].length;
        const urlNode = node.splitText(start);
        const rest = urlNode.splitText(m[0].length);
        const a = document.createElement('a');
        a.href = /^www\./i.test(m[0]) ? 'https://' + m[0] : m[0];
        urlNode.replaceWith(a);
        a.appendChild(urlNode);
        // Cursor no texto depois do link (fora do <a>, para o que vier a seguir não entrar no link)
        const range = document.createRange();
        range.setStart(rest, 0);
        range.collapse(true);
        sel.removeAllRanges();
        sel.addRange(range);
        syncBody();
      }
    }

    function openTaskPicker(note, bar) {

      const update = () => {
        const words = foldText(input.value).split(/\s+/).filter(Boolean);

        matches = allTaskList()
          .filter(t => t.uid && !note.tasks.includes(String(t.uid)))
          .filter(t => { const hay = foldText('#' + t.uid + ' ' + t.title); return words.every(w => hay.includes(w)); })
          .sort((a, b) => (a.checked - b.checked) || ((taskScore(b) ?? -1) - (taskScore(a) ?? -1)))
          .slice(0, 10);
        sel = 0;
        results.style.display = matches.length ? '' : 'none';
        results.innerHTML = matches.map((t, k) => `<div class="${k === sel ? 'sel' : ''}${t.checked ? ' done' : ''}" style="--cat-color:${categoryColor(taskCategory(t))}"><small>#${escapeHtml(t.uid)}</small>${escapeHtml(t.title)}</div>`).join('');
        [...results.children].forEach((el, k) => el.addEventListener('mousedown', e => { e.preventDefault(); pick(matches[k]); }));
      };
      input.addEventListener('input', update);
      input.addEventListener('keydown', e => {
        if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); close(); }
        else if (e.key === 'ArrowDown' || e.key === 'ArrowUp') {
          e.preventDefault();

          [...results.children].forEach((el, k) => el.classList.toggle('sel', k === sel));
        } else if (e.key === 'Enter') { e.preventDefault(); if (matches[sel]) pick(matches[sel]); }
      });
      input.addEventListener('blur', () => setTimeout(() => { if (picker.isConnected) close(); }, 150));
      update();
      input.focus();
    }
